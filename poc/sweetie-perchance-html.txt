function lineOfSight(ax,ay,bx,by,g){ g=g||GRID; var steps=Math.ceil(Math.hypot(bx-ax,by-ay)/(CELL*0.5));
  for(var sIx=0;sIx<=steps;sIx++){ var t=steps?sIx/steps:0, x=ax+(bx-ax)*t, y=ay+(by-ay)*t; var c=cellOf(x,y); if(!passable(c[0],c[1],g))return false; } return true; }

function steerAvoid(x,y,gx,gy,obs,cfg){ cfg=cfg||{lookahead:1.2,clearance:ROBOT_R+0.04,pass_margin:0.12,arrival:0.5,speed_floor:0.45};

// ==== SAFETY GUARD ==== a hard gate BELOW cognition. Limits are fixed constants; no tool,
// chat message, or memory can change or disable them (the Go2 lesson: never let the LLM talk
// its own safety off). Scales speed by proximity, with extra clearance for living things and
// a hard cap when moving toward UNSENSED (blind-zone) space -- the LiDAR-blind-spot lesson.
var SG={ HARD:0.18, SOFT_HARD:0.45, RAMP:0.6, FOV:1.4, BLINDCAP:0.45 };
var safety={ state:"clear", scale:1, blocked:0, lastBlocked:"" };
function capFor(edge, soft, blind){ var hard=soft?SG.SOFT_HARD:SG.HARD, slow=hard+SG.RAMP, sc;
  if(edge<=hard) sc=0; else if(edge<slow) sc=(edge-hard)/(slow-hard); else sc=1;
  if(blind && edge<slow*1.5) sc=Math.min(sc, SG.BLINDCAP); return sc; }
function safetyScale(){ var r=world.robot, worst=1, state="clear";
  var ents=[]; for(var i=0;i<world.obstacles.length;i++){ var o=world.obstacles[i]; ents.push({x:o.x,y:o.y,r:o.r,soft:false}); }
  ents.push({x:world.pet.x,y:world.pet.y,r:world.pet.r,soft:true}); ents.push({x:world.user.x,y:world.user.y,r:0.18,soft:true});
  for(var k=0;k<ents.length;k++){ var e=ents[k], dx=e.x-r.x, dy=e.y-r.y, dist=Math.hypot(dx,dy), edge=dist-e.r-ROBOT_R;
    var rel=wrap(Math.atan2(dy,dx)-r.yaw), ahead=Math.cos(rel);
    // Direction-aware: only brake for things in the PATH of travel (ahead of the heading). A thing
    // behind the heading is not in the way and must NEVER freeze her -- otherwise she gets pinned
    // against a wall / the pet / you and cannot drive away. Still guarantees "never drive INTO things".
    if(ahead<=0.15) continue;
    var blind=Math.abs(rel)>SG.FOV, sc=capFor(edge,e.soft,blind); sc=1-(1-sc)*ahead;
    if(sc<worst){ worst=sc; state = sc<=0.02 ? "STOP" : (blind?"blind-slow":"slowing"); } }
  if(worst>=0.999) state="clear"; safety.scale=worst; safety.state=state; return worst; }

// ---- energy-aware goal arbiter -------------------------------------------------------------
// The fix for "seeks a goal, ignores low battery": every goal is scored by how much it's worth
// vs what it costs to travel there, and a hard RETURN-TO-HOME reserve preempts ANYTHING (even a
// user task) once remaining charge is only just enough to reach the charger. Safety over wants.
function pathLen(ax,ay,bx,by){ var raw=aStar(ax,ay,bx,by,GRID); if(!raw)return Infinity;
  var sm=smoothPath(raw,GRID),d=0; for(var i=1;i<sm.length;i++){ d+=Math.hypot(sm[i].x-sm[i-1].x,sm[i].y-sm[i-1].y); } return d; }
